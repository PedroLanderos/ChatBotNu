# -*- coding: utf-8 -*-
import re
import unicodedata

def normaliza(txt: str) -> str:
    txt = (txt or "").strip()
    txt = unicodedata.normalize("NFD", txt)
    txt = "".join(ch for ch in txt if unicodedata.category(ch) != "Mn")
    return txt.lower()
